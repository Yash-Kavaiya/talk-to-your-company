\section{The problem}

\begin{frame}{Nobody can watch every camera}
  \begin{minipage}[t]{0.235\textwidth}
    \triadcardbox[equal height group=problem]{\faIcon{eye-slash}}{Events go unseen}{Near-misses, restricted zones and missing helmets slip by.}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.235\textwidth}
    \triadcardbox[equal height group=problem]{\faIcon{walking}}{Answers are slow}{``What is happening on floor 3?'' means walking there.}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.235\textwidth}
    \triadcardbox[equal height group=problem]{\faIcon{video}}{Walls are mute}{Camera walls show video. They cannot be asked anything.}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.235\textwidth}
    \triadcardbox[equal height group=problem]{\faIcon{cloud}}{Cloud is a poor fit}{Factory video off site is slow, costly and often not allowed.}
  \end{minipage}
  \vskip2.6ex
  {\large\bfseries The manager needs \triadhl{one view of the whole site}, and \triadhl{a way to ask it things}.\par}
\end{frame}

\begin{frame}[plain,noframenumbering]
  \triadstatement[Ask by voice. The twin flies to the right floor, shows what the cameras see, and answers out loud.]
    {A live 3D twin of your company that you can talk to.}
\end{frame}
