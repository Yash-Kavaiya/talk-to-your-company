\section{Architecture in detail}
\renewcommand{\arraystretch}{1.3}

\begin{frame}{One process runs three loops at once}
  \begin{minipage}[t]{0.32\textwidth}
    \triadcardbox[equal height group=loops]{\faIcon{video}}{Perception threads}{One thread per stream decodes video. One batched detector, then tracker and mapper.}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.32\textwidth}
    \triadcardbox[equal height group=loops]{\faIcon{broadcast-tower}}{Broadcast loop, 5 Hz}{Runs the rules on the latest tracks, stores events, sends state to the browser.}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.32\textwidth}
    \triadcardbox[equal height group=loops]{\faIcon{comments}}{Agent loop, per question}{Speech to text, up to three rounds of tool calls, then the spoken answer.}
  \end{minipage}
  \vskip2.4ex
  {\bfseries One FastAPI process on \triadhl{0.0.0.0:8000}. The browser is the only client; that is the only port.\par}
\end{frame}

\begin{frame}{Perception: five stages from pixels to floor metres}
  \footnotesize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.12\textwidth}>{\raggedright\arraybackslash}p{0.17\textwidth}>{\raggedright\arraybackslash}p{0.66\textwidth}@{}}
    {\scriptsize\bfseries\color{triadPrimary}STAGE} & {\scriptsize\bfseries\color{triadPrimary}MODULE} & {\scriptsize\bfseries\color{triadPrimary}WHAT IT DOES}\\[1.0ex]
    \textbf{Ingest} & \texttt{ingest.py} & \textcolor{triadMuted}{GStreamer hardware decode, OpenCV fallback. Files, a webcam or shared frames.}\\[0.7ex]
    \textbf{Detect} & \texttt{detector.py} & \textcolor{triadMuted}{YOLO on TensorRT FP16. Every stream's latest frame in one batch.}\\[0.7ex]
    \textbf{Track} & \texttt{tracker.py} & \textcolor{triadMuted}{ByteTrack-style association per stream. Gives each person a stable id.}\\[0.7ex]
    \textbf{Map} & \texttt{mapper.py} & \textcolor{triadMuted}{Foot point through the camera homography to floor metres.}\\[0.7ex]
    \textbf{Mock} & \texttt{mock.py} & \textcolor{triadMuted}{Simulated people; a scripted scenario every 30 s raises a real event.}\\
  \end{tabular}
\end{frame}

\begin{frame}{Rules are pure functions with fixed thresholds}
  \footnotesize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.20\textwidth}>{\raggedright\arraybackslash}p{0.42\textwidth}>{\raggedright\arraybackslash}p{0.32\textwidth}@{}}
    {\scriptsize\bfseries\color{triadPrimary}EVENT} & {\scriptsize\bfseries\color{triadPrimary}RAISED WHEN} & {\scriptsize\bfseries\color{triadPrimary}NOTES}\\[1.0ex]
    \texttt{near\_miss} & Person and forklift within 2 m, closing above 0.5 m/s. & \textcolor{triadMuted}{Severity 3.}\\[0.7ex]
    \texttt{restricted\_zone} & Person inside a restricted polygon for over 2 s. & \textcolor{triadMuted}{Once per stay; authorised people exempt.}\\[0.7ex]
    \texttt{no\_helmet} & The vision model says no about a person crop. & \textcolor{triadMuted}{One unchecked person every 3 s.}\\[0.7ex]
    \texttt{crowding} & More than 4 people within a 3 m radius. & \textcolor{triadMuted}{Severity 1.}\\
  \end{tabular}
  \vskip1.6ex
  {\bfseries Unit-tested, debounced for \triadhl{30 seconds}, and the same on mock and real tracks.\par}
\end{frame}

    \texttt{focus\_view} & Nothing. & \textcolor{triadMuted}{Flies the camera there.}\\[0.45ex]
    \texttt{show\_event} & Nothing. & \textcolor{triadMuted}{Opens the event panel.}\\[0.45ex]
    \texttt{identify\_person} & Employee, visit timing, incidents. & \textcolor{triadMuted}{Opens the person panel.}\\[0.45ex]

\begin{frame}{Known risks, and what is already in place for each}
  \footnotesize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.38\textwidth}>{\raggedright\arraybackslash}p{0.56\textwidth}@{}}
    {\scriptsize\bfseries\color{triadPrimary}RISK} & {\scriptsize\bfseries\color{triadPrimary}MITIGATION}\\[1.0ex]
    The shared GPU slows everything. & \textcolor{triadMuted}{Detector at 10 fps, models pre-warmed, a backup video.}\\[0.7ex]
    A dependency replaces torch. & \textcolor{triadMuted}{Every install is dry-run checked first.}\\[0.7ex]
    A small model is weak at tool calling. & \textcolor{triadMuted}{JSON-constrained calls; keyword router as fallback.}\\[0.7ex]
    The microphone is blocked in the browser. & \textcolor{triadMuted}{The text box does the same thing.}\\[0.7ex]
    A model fails to load on stage. & \textcolor{triadMuted}{That backend starts as its mock.}\\
  \end{tabular}
\end{frame}
